Linux commands for exploring a running system

ps aux                 # all processes          top / htop    # live CPU/memory by process
pstree -p              # process tree           kill -TERM <pid>   # ask a process to exit
free -h                # memory and swap        vmstat 1      # memory, paging, CPU each second
iostat -x 1            # disk utilization       lsof -p <pid>   # open files of a process
cat /proc/<pid>/status # PCB-like details      cat /proc/meminfo
ls -li                 # show inode numbers     stat file     # inode metadata
df -h  /  du -sh .     # file system and directory usage
strace -c ls           # count system calls made by ls
ulimit -a              # per-process resource limits

Python: inspecting the system

import os, psutil          # pip install psutil

print(os.getpid(), os.getppid())          # this process and its parent
p = psutil.Process()
print(p.status(), p.num_threads(), p.memory_info().rss)

print(psutil.cpu_count(logical=True), psutil.cpu_percent(interval=1, percpu=True))
vm = psutil.virtual_memory();  print(vm.total, vm.available, vm.percent)
print(psutil.swap_memory())
print(psutil.disk_usage("/"))
for proc in psutil.process_iter(["pid", "name", "cpu_percent"]):
    print(proc.info)

Python: threads and a lock

import threading
counter, lock = 0, threading.Lock()

def work(n):
    global counter
    for _ in range(n):
        with lock:          # remove the lock and run with large n to see lost updates
            counter += 1

ts = [threading.Thread(target=work, args=(100_000,)) for _ in range(4)]
[t.start() for t in ts]; [t.join() for t in ts]
print(counter)             # 400000 with the lock

Note: in some CPython versions the GIL hides many races for simple operations. The lock is still required for correctness.
